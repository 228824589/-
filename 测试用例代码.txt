{
    // 诊断请求字符串定义
    char send_string_1003[30] = "0x1003";
    
    // 安全访问请求 - 请求种子
    char send_string_2701[30] = "0x2701";
    char send_string_2703[30] = "0x2703"; 
    char send_string_2721[30] = "0x2721";
    
    // 安全访问请求 - 发送密钥（错误密钥）
    char send_string_2702_wrong[30] = "0x270201020304";  // 错误密钥示例
    char send_string_2704_wrong[30] = "0x270405060708";  // 错误密钥示例
    char send_string_2722_wrong[30] = "0x2722090A0B0C";  // 错误密钥示例
    
    // 预期响应字符串
    char resp_5003[30] = "5003";
    char resp_6701[30] = "6701";

    char resp_67xx[30] = "67";           // 通用肯定响应前缀

    struct T2DiagCDD_Response diagResp;
    enum T2Log_Display log_flag = g_T2LOG_DISPLAY_TRUE;

                                 "SecurityAccessValidation", "安全访问验证测试");

    /******************** 第一轮安全访问测试 (27 01/02) ********************/


@Simulation::Sim All =0;
 diagStopTesterPresent("ccu");
NetworkSleepCheck（180）;//单位s,休眠后唤醒
@Simulation::Sim_All =1；
testWaitForTimeout(6000)；

    teststep("S1", "发送1003请求 - 进入扩展诊断会话");
    T2DiagCDD_Service(diagResp, send_string_1003, log_flag);
    compareResponse(diagResp, 6, resp_5003, COMPARE_FIRST_BYTES, 2);

    teststep("S2", "发送2701请求 - 安全访问请求种子");
    T2DiagCDD_Service(diagResp, send_string_2701, log_flag);
    compareResponse(diagResp, 6, resp_67xx, COMPARE_FIRST_BYTES, 2);

    teststep("S3", "发送2702请求 - 发送错误密钥");
    T2DiagCDD_Service(diagResp, send_string_2702_wrong, log_flag);
    compareResponse(diagResp, 3, resp_7F2735, COMPARE_FULL, NOT_USED);

    teststep("S4", "重新建立会话并尝试解锁（应被拒绝）");
    // 先建立会话
    T2DiagCDD_Service(diagResp, send_string_1003, log_flag);
    compareResponse(diagResp, 6, resp_5003, COMPARE_FIRST_BYTES, 2);
    使用解锁函数发送正确密钥，但由于种子已失效，应该被拒绝,这里需要获取失败的原因.
    T2DiagCDD_Service27_Unlock(diagResp, 0x01, log_flag);
compareResponse(diagResp,3, resp_7F2724, COMPARE_FIRST_BYTES, 3);   //检查是否返回24,
    // 解锁函数会自动报告失败到测试报告

    /******************** 第二轮安全访问测试 (27 03/04) ********************/
    teststep("S5", "发送1003请求 - 重新建立诊断会话");
    T2DiagCDD_Service(diagResp, send_string_1003, log_flag);
    compareResponse(diagResp, 6, resp_5003, COMPARE_FIRST_BYTES, 2);

    teststep("S6", "发送2703请求 - 安全访问请求新种子");
    T2DiagCDD_Service(diagResp, send_string_2703, log_flag);
    compareResponse(diagResp, 6, resp_67xx, COMPARE_FIRST_BYTES, 2);

    teststep("S7", "发送2704请求 - 发送错误密钥");
    T2DiagCDD_Service(diagResp, send_string_2704_wrong, log_flag);
    compareResponse(diagResp, 3, resp_7F2735, COMPARE_FULL, NOT_USED);

    teststep("S8", "重新建立会话并尝试解锁（应被拒绝）");
    // 先建立会话
    T2DiagCDD_Service(diagResp, send_string_1003, log_flag);
    compareResponse(diagResp, 6, resp_5003, COMPARE_FIRST_BYTES, 2);
    // 使用解锁函数发送正确密钥
    T2DiagCDD_Service27_Unlock(diagResp, 0x03, log_flag);
    // 解锁函数会自动报告失败

    teststep("S9", "发送1003请求 - 重新建立诊断会话");
    T2DiagCDD_Service(diagResp, send_string_1003, log_flag);

    compareResponse(diagResp, 6, resp_67xx, COMPARE_FIRST_BYTES, 2);

    teststep("S11", "发送2722请求 - 发送错误密钥");

    // 使用解锁函数发送正确密钥
    T2DiagCDD_Service27_Unlock(diagResp, 0x21, log_flag);

    teststep("S13", "验证正常解锁流程 - 发送1003请求");
    T2DiagCDD_Service(diagResp, send_string_1003, log_flag);
    compareResponse(diagResp, 6, resp_5003, COMPARE_FIRST_BYTES, 2);

    teststep("S14", "验证正常解锁流程 - 使用解锁函数（应成功）");
    T2DiagCDD_Service27_Unlock(diagResp, 0x01, log_flag);
    // 解锁函数会自动报告成功

    /******************** 测试总结 ********************/
    teststep("SUMMARY", "安全访问种子失效机制验证完成");
    teststep("SUMMARY", "验证结论：错误密钥会导致种子立即失效");

    T2Log_WriteTestCaseVerdictToLog(g_T2FUNCTION_LOG_FILE_HANDLE, module_name);
    T2Log_SetLogFileName_StopLogging();

2.制造IBCS节点丢失后,无法读取速度故障码
testcase DSTC509()
{
    // 诊断请求字符串定义
    char send_string_14FFFFFF[30] = "0x14FFFFFF";
    char send_string_190209[30] = "0x190209";  // 19服务02子功能 - 读取当前故障码
    
    // 预期响应字符串
    char resp_positive[30] = "54";           // 肯定响应前缀
    char resp_5902[30] = "5902";             // 19服务肯定响应
    
    // DTC定义
    byte DTC_CommLoss[3] = {0xXX, 0xXX, 0xXX};        // 通信丢失DTC（需要根据实际情况填写）
    byte DTC_C41586[3] = {0xC4, 0x15, 0x86};          // 速度故障DTC
    
    // 诊断变量
    struct T2DiagCDD_Response diagResp;
    struct T2DiagCDD_DTC DTC_list[2000];
    int DTC_num;
    enum T2Log_Display log_flag = g_T2LOG_DISPLAY_TRUE;
    enum T2Log_Display not_log_flag = g_T2LOG_DISPLAY_FALSE;
    
    byte desDTC[8192];
    long deslength;
    
    // 开始日志记录
    T2Log_SetLogFileName_StartLogging(setting_log_final_path, "DSTC509");
    T2Log_WriteTestCaseTitleToLog(g_T2FUNCTION_LOG_FILE_HANDLE, module_name, 
                                 "DSTC509", "网络故障设置逻辑测试");

    /******************** 步骤1: 唤醒BCU并清除诊断信息 ********************/

    teststep("1", "唤醒BCU并清除诊断信息");
    /******************** 总线休眠 ********************/
    teststep("SLEEP", "总线休眠");
    Simulation::Sim_All = 0;
    diagStopTesterPresent("ccu");
    NetworkSleepCheck(180);  // 休眠180秒
    Simulation::Sim_All = 1;
    testWaitForTimeout(6000);// 唤醒后等待6秒

    
    // 清除诊断信息
    T2DiagCDD_Service(diagResp, send_string_14FFFFFF, log_flag);
    compareResponse(diagResp, NOT_USED, resp_positive, COMPARE_EVENT_CODE, positive);

    /******************** 步骤2: 初始状态验证 ********************/
    teststep("2", "发送190209读取DTC，确认无网络相关故障码");
    T2DiagCDD_Service19_SubFunctionA(diagResp, 0x02, DTC_list, DTC_num, not_log_flag);  // 使用02子功能
    
    // 复制响应数据用于DTC检查
    T2Function_clearArray(desDTC);
    memcpy(desDTC, diagResp.respData, diagResp.length);
    deslength = diagResp.length;
    
    // 验证没有通信丢失和速度故障DTC
    T2DiagDD_CheckDTCAndStatusInResponse(desDTC, deslength, DTC_CommLoss, NOT_USED, DTC_EXPECT_ABSENT);
    T2DiagDD_CheckDTCAndStatusInResponse(desDTC, deslength, DTC_C41586, NOT_USED, DTC_EXPECT_ABSENT);

    /******************** 步骤3: 模拟节点丢失通信 ********************/
    teststep("3", "模拟IBCS节点丢失通讯");
    disablenode(ibcs);

    /******************** 步骤4: 验证通信丢失DTC ********************/
    teststep("4", "发送190209读取节点丢失通信的当前DTC");
    T2DiagCDD_Service19_SubFunctionA(diagResp, 0x02, DTC_list, DTC_num, not_log_flag);  // 使用02子功能
    
    T2Function_clearArray(desDTC);
    memcpy(desDTC, diagResp.respData, diagResp.length);
    deslength = diagResp.length;
    
    // 验证通信丢失DTC为当前故障（使用0x2F）
    T2DiagDD_CheckDTCAndStatusInResponse(desDTC, deslength, DTC_CommLoss, 0x2F, DTC_EXPECT_PRESENT);

    /******************** 步骤5: 模拟速度故障 ********************/
    teststep("5", "模拟IBCS节点的速度故障");
    $PCAN::IBCS::BCS_2_C::BCS_VehspdVD = 0;

    /******************** 步骤6: 验证故障状态 ********************/
    teststep("6", "发送190209验证只有通信丢失当前DTC，无速度故障DTC");
    T2DiagCDD_Service19_SubFunctionA(diagResp, 0x02, DTC_list, DTC_num, not_log_flag);  // 使用02子功能
    
    T2Function_clearArray(desDTC);
    memcpy(desDTC, diagResp.respData, diagResp.length);
    deslength = diagResp.length;
    
    // 验证通信丢失DTC仍为当前故障（使用0x2F）
    T2DiagDD_CheckDTCAndStatusInResponse(desDTC, deslength, DTC_CommLoss, 0x2F, DTC_EXPECT_PRESENT);
    // 验证速度故障DTC不存在
    T2DiagDD_CheckDTCAndStatusInResponse(desDTC, deslength, DTC_C41586, NOT_USED, DTC_EXPECT_ABSENT);

    /******************** 步骤7: 恢复节点通信 ********************/
    teststep("7", "恢复IBCS节点通信");
    enablenode(ibcs);

    /******************** 步骤8: 验证故障状态转换 ********************/
    teststep("8", "发送190209读取故障状态");
    T2DiagCDD_Service19_SubFunctionA(diagResp, 0x02, DTC_list, DTC_num, not_log_flag);  // 使用02子功能
    
    T2Function_clearArray(desDTC);
    memcpy(desDTC, diagResp.respData, diagResp.length);
    deslength = diagResp.length;
    
    // 验证通信丢失变为历史故障（使用0x2E），速度故障为当前故障（使用0x2F）
    T2DiagDD_CheckDTCAndStatusInResponse(desDTC, deslength, DTC_CommLoss, 0x2E, DTC_EXPECT_PRESENT);
    T2DiagDD_CheckDTCAndStatusInResponse(desDTC, deslength, DTC_C41586, 0x2F, DTC_EXPECT_PRESENT);

    /******************** 步骤9: 再次模拟通信丢失 ********************/
    teststep("9", "再次模拟IBCS节点丢失通信故障");
    disablenode(ibcs);

    /******************** 步骤10: 最终验证 ********************/
    teststep("10", "发送190209读取最终故障状态");
    T2DiagCDD_Service19_SubFunctionA(diagResp, 0x02, DTC_list, DTC_num, not_log_flag);  // 使用02子功能
    
    T2Function_clearArray(desDTC);
    memcpy(desDTC, diagResp.respData, diagResp.length);
    deslength = diagResp.length;
    
    // 验证通信丢失为当前故障（使用0x2F），速度故障为历史故障（使用0x2E）
    T2DiagDD_CheckDTCAndStatusInResponse(desDTC, deslength, DTC_CommLoss, 0x2F, DTC_EXPECT_PRESENT);
    T2DiagDD_CheckDTCAndStatusInResponse(desDTC, deslength, DTC_C41586, 0x2E, DTC_EXPECT_PRESENT);

    /******************** 清理步骤: 恢复所有设置 ********************/
    teststep("CLEANUP", "恢复所有设置");
    // 恢复速度信号
    $PCAN::IBCS::BCS_2_C::BCS_VehspdVD = 1;
    // 恢复节点通信
    enablenode(ibcs);
    // 清除诊断信息
    T2DiagCDD_Service(diagResp, send_string_14FFFFFF, log_flag);
    


    /******************** 测试总结 ********************/
    teststep("SUMMARY", "网络故障设置逻辑验证完成");
    teststep("SUMMARY", "验证结论：通信丢失故障优先于速度故障");
    teststep("SUMMARY", "故障状态转换逻辑符合预期");

    // 结束日志记录
    T2Log_WriteTestCaseVerdictToLog(g_T2FUNCTION_LOG_FILE_HANDLE, module_name);
    T2Log_SetLogFileName_StopLogging();
}
