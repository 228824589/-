/**
 * 复制诊断响应数据到目标数组
 * @param diagResp - 诊断响应结构体（值传递：复制整个结构体）
 * @param targetArray - 目标数组（引用传递：修改会同步到原始数组）
 * @param targetLength - 目标数组长度（引用传递：修改会同步到原始变量）
 */
void copyDiagnosticResponse(struct T2DiagCDD_Response diagResp, byte targetArray[], long &targetLength)
{
    T2Function_clearArray(targetArray);
    memcpy(targetArray, diagResp.respData, diagResp.length);
    targetLength = diagResp.length;
}
